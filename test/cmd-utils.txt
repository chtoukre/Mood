CREATE TABLE mood_entries (
    date DATE PRIMARY KEY,
    moral_global INTEGER,
    gratitude INTEGER,
    pleine_conscience INTEGER,
    famille INTEGER,
    amis INTEGER,
    relation INTEGER,
    plaisir INTEGER,
    calme INTEGER,
    sincerite INTEGER,
    temps_pour_soi INTEGER,
    alimentation INTEGER,
    hydratation INTEGER,
    exercice INTEGER,
    sorties INTEGER,
    sante INTEGER,
    creativite INTEGER,
    finances INTEGER,
    education_travail INTEGER,
    emotions_pensees INTEGER,
    present INTEGER,
    futur INTEGER,
    overall_mood INTEGER,
    note TEXT
);

SELECT
    date,
    (
        (
            gratitude + pleine_conscience + famille + amis +
            relation + plaisir + calme + sincerite + temps_pour_soi +
            alimentation + hydratation + exercice + sorties + sante +
            creativite + finances + education_travail +
            emotions_pensees + present + futur
        ) / 20.0
    ) AS moyenne_autres_aspects
FROM mood_entries
ORDER BY date ASC;
